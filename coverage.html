func main() {
        cfg, err := config.Parse("config.yaml")
        if err != nil {
                slog.Error("parse config", slog.String("error", err.Error()))
                return
        }

        log := mustMakeLogger(cfg.Logger)

        if err := run(cfg, log); err != nil {
                log.Error("running app", slog.String("error", err.Error()))
        }
}

func run(cfg config.Config, log *slog.Logger) error {
        signalChan := make(chan os.Signal, 1)
        signal.Notify(signalChan, syscall.SIGINT, syscall.SIGTERM)

        evRepo := &mockEvRepo{log}
        batcherOpts := makeBatcherOptions(cfg.Batcher)
        batcherOpts = append(batcherOpts, batcher.WithLogger(log.With(slog.String("service", "batcher"))))
        b := batcher.New(evRepo, batcherOpts...)
        go b.Start()

        procs := []pipeline.Processor{new(processors.Validate)}
        p := pipeline.New(procs, b.Add, pipeline.WithLogger(log.With(slog.String("service", "pipeline"))), pipeline.WithWorkerCount(4))
        go p.Start()

        s := server.New(cfg.HTTPServer.Address, p, log.With(slog.String("service", "http server")))
        go func() {
                if err := s.Start(); err != nil {
                        log.Error("run http server", slog.String("error", err.Error()))
                }
        }()

        <-signalChan

        ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
        defer cancel()

        if err := s.Shutdown(ctx); err != nil {
                return fmt.Errorf("shutdown server: %w", err)
        }

        if err := p.Shutdown(ctx); err != nil {
                return fmt.Errorf("shutdown pipeline: %w", err)
        }

        if err := b.Shutdown(ctx); err != nil {
                return fmt.Errorf("shutdown batcher: %w", err)
        }

        return nil
}

func mustMakeLogger(cfg config.LoggerConfig) *slog.Logger {
        var level slog.Level
        switch cfg.Level {
        case "DEBUG":
                level = slog.LevelDebug
        case "INFO":
                level = slog.LevelInfo
        case "WARN":
                level = slog.LevelWarn
        case "ERROR":
                level = slog.LevelError
        default:
                level = slog.LevelInfo
        }

        opts := slog.HandlerOptions{
                AddSource: cfg.AddSource,
                Level:     level,
        }
        handler := slog.NewJSONHandler(os.Stdout, &opts)
        return slog.New(handler)
}

func makeBatcherOptions(cfg config.BatcherConfig) []batcher.Option {
        var opts []batcher.Option
        if cfg.BatchSize != 0 {
                opts = append(opts, batcher.WithBatchSize(cfg.BatchSize))
        }
        if cfg.EventsBufferSize != 0 {
                opts = append(opts, batcher.WithEventsChanBufferSize(cfg.EventsBufferSize))
        }
        if cfg.FlushChannelSize != 0 {
                opts = append(opts, batcher.WithFlushChanBufferSize(cfg.FlushChannelSize))
        }
        if cfg.WorkerCount != 0 {
                opts = append(opts, batcher.WithWorkerCount(cfg.WorkerCount))
        }
        if cfg.SendTickerDuration != 0 {
                opts = append(opts, batcher.WithSendTickerDuration(cfg.SendTickerDuration))
        }
        if cfg.WorkerTimeout != 0 {
                opts = append(opts, batcher.WithWorkerTimeout(cfg.WorkerTimeout))
        }

        return opts
}

var (
        ErrShuttingDown    = errors.New("shutting down")
        ErrAlreadyStoppped = errors.New("already stopped")
)

type Option func(*Batcher)

type EventRepository interface {
        CreateEvents(ctx context.Context, events []event.Event) error
}

type Batcher struct {
        eventsChSize            int
        flushChSize             int
        batchSendTickerDuration time.Duration

        workerCount   int
        workerTimeout time.Duration
        batchSize     int

        mu       sync.RWMutex
        eventsCh chan event.Event

        flushCh chan []event.Event
        evRepo  EventRepository

        stopCtx    context.Context
        stopCancel context.CancelFunc
        closeOnce  sync.Once
        closing    chan struct{}
        done       chan struct{}

        log *slog.Logger
}

func New(evRepo EventRepository, opts ...Option) *Batcher {
        stopCtx, stopCancel := context.WithCancel(context.Background())

        b := &Batcher{
                eventsChSize:            DefaultEventsBufferSize,
                flushChSize:             DefaultFlushChanSize,
                batchSendTickerDuration: DefaultBatchSendTickerDuration,
                workerCount:             DefaultWorkerCount,
                workerTimeout:           DefaultWorkerTimeout,
                batchSize:               DefaultBatchSize,
                evRepo:                  evRepo,
                stopCtx:                 stopCtx,
                stopCancel:              stopCancel,
                closing:                 make(chan struct{}),
                done:                    make(chan struct{}),
                log:                     slog.New(slog.DiscardHandler),
        }

        for _, opt := range opts {
                opt(b)
        }

        b.eventsCh = make(chan event.Event, b.eventsChSize)
        b.flushCh = make(chan []event.Event, b.flushChSize)

        return b
}

func (b *Batcher) Add(ctx context.Context, ev event.Event) error {
        b.mu.RLock()
        defer b.mu.RUnlock()

        select {
        case <-b.closing:
                return ErrShuttingDown
        default:
        }

        select {
        case <-ctx.Done():
                return ctx.Err()
        case b.eventsCh <- ev:
                return nil
        case <-b.closing:
                return ErrShuttingDown
        }
}

func (b *Batcher) Start() {
        defer func() {
                b.stopCancel()
                close(b.done)

                b.log.Info("batcher stopped")
        }()
        b.log.Info("batcher started")

        var workersWg sync.WaitGroup
        workersWg.Go(b.startWorkers)
        b.startLoop()
        close(b.flushCh)
        workersWg.Wait()
}

func (b *Batcher) Shutdown(ctx context.Context) error {
        alreadyStopped := true
        b.closeOnce.Do(func() {
                alreadyStopped = false
                close(b.closing)
        })
        if alreadyStopped {
                return ErrAlreadyStoppped
        }

        b.mu.Lock()
        close(b.eventsCh)
        b.mu.Unlock()

        select {
        case <-ctx.Done():
                b.stopCancel()
                return ctx.Err()
        case <-b.done:
                return nil
        }
}

func (b *Batcher) startLoop() {
        ticker := time.NewTicker(b.batchSendTickerDuration)
        defer ticker.Stop()

        batch := make([]event.Event, 0, b.batchSize)
        for {
                select {

                case <-b.stopCtx.Done():
                        if n := len(b.eventsCh) + len(batch); n > 0 {
                                b.log.Warn("events left in batcher", slog.Int("count", n))
                        }
                        return

                case ev, ok := <-b.eventsCh:
                        if !ok {
                                b.flush(batch)
                                b.log.Info("events drained")
                                return
                        }

                        if b.stopCtx.Err() != nil {
                                b.log.Warn("events left in batcher", slog.Int("count", len(b.eventsCh)+len(batch)+1))
                                return
                        }

                        batch = append(batch, ev)

                        if len(batch) == b.batchSize {
                                batch = b.flush(batch)
                                ticker.Reset(b.batchSendTickerDuration)
                        }

                case <-ticker.C:
                        batch = b.flush(batch)
                }
        }
}

func (b *Batcher) flush(batch []event.Event) []event.Event {
        if len(batch) == 0 {
                return batch
        }
        select {
        case b.flushCh <- batch:
                return make([]event.Event, 0, b.batchSize)
        case <-b.stopCtx.Done():
                b.log.Warn("batch dropped on shutdown", slog.Int("count", len(batch)))
                return nil
        }
}

func (b *Batcher) startWorkers() {
        var wg sync.WaitGroup
        for range b.workerCount {
                wg.Go(func() {
                        for {
                                select {

                                case <-b.stopCtx.Done():
                                        if n := len(b.flushCh); n > 0 {
                                                b.log.Warn("batches left in bathcer", slog.Int("count", n))
                                        }
                                        return

                                case batch, ok := <-b.flushCh:
                                        if !ok {
                                                b.log.Info("batches drained")
                                                return
                                        }

                                        if b.stopCtx.Err() != nil {
                                                b.log.Warn("bathces left in batcher", slog.Int("count", len(b.flushCh)+1))
                                                return
                                        }

                                        workCtx, workCancel := context.WithTimeout(b.stopCtx, b.workerTimeout)
                                        err := b.evRepo.CreateEvents(workCtx, batch)
                                        workCancel()
                                        if err != nil {
                                                b.log.Error("save events in repository", slog.String("error", err.Error()))
                                        }

                                }
                        }
                })
        }

        wg.Wait()
}

package cache

import (
        "errors"
        "hash/maphash"
        "sync"
)

type shard[K comparable, V any] struct {
        sync.RWMutex
        data map[K]V
}

type ShardedCache[K comparable, V any] struct {
        shards []*shard[K, V]
        hasher func(K) uint64
}

func NewSharded[K comparable, V any](shardCount int) (*ShardedCache[K, V], error) {
        if shardCount <= 0 {
                return nil, errors.New("shard count must be positive")
        }

        shards := make([]*shard[K, V], shardCount)
        for i := range shards {
                shards[i] = &shard[K, V]{
                        data: make(map[K]V),
                }
        }

        seed := maphash.MakeSeed()
        hasher := func(key K) uint64 {
                return maphash.Comparable(seed, key)
        }

        return &ShardedCache[K, V]{
                shards: shards,
                hasher: hasher,
        }, nil
}

func (c *ShardedCache[K, V]) Get(key K) (V, bool) {
        hash := c.hasher(key)

        return c.shards[hash%uint64(len(c.shards))].get(key)
}

func (c *ShardedCache[K, V]) Put(key K, val V) {
        hash := c.hasher(key)

        c.shards[hash%uint64(len(c.shards))].put(key, val)
}

func (c *ShardedCache[K, V]) Delete(key K) {
        hash := c.hasher(key)

        c.shards[hash%uint64(len(c.shards))].delete(key)
}

func (s *shard[K, V]) get(key K) (V, bool) {
        s.RLock()
        defer s.RUnlock()
        val, ok := s.data[key]
        return val, ok
}

func (s *shard[K, V]) put(key K, val V) {
        s.Lock()
        defer s.Unlock()
        s.data[key] = val
}

func (s *shard[K, V]) delete(key K) {
        s.Lock()
        defer s.Unlock()
        delete(s.data, key)
}

package pipeline

import (
        "context"
        "errors"
        "fmt"
        "log/slog"
        "sync"
        "time"

        "github.com/neelalala/hopper/internal/event"
)

const (
        DefaultSinkTimeout   = 10 * time.Second
        DefaultWorkerTimeout = 30 * time.Second
        DefaultBufferSize    = 10_000
        DefaultWorkerCount   = 16
)

var (
        ErrChanIsFull     = errors.New("chan is full")
        ErrShuttingDown   = errors.New("shutting down")
        ErrAlreadyStopped = errors.New("already stopped")
)

type Option func(p *Pipeline)

type Processor interface {
        Process(context.Context, *event.Event) (drop bool, err error)
}

type Sink func(context.Context, event.Event) error

type Pipeline struct {
        eventsBufferSize int

        processors  []Processor
        sink        Sink
        sinkTimeout time.Duration

        workerCount   int
        workerTimeout time.Duration

        mu     sync.RWMutex
        closed bool
        events chan event.Event

        stopCtx    context.Context
        stopCancel context.CancelFunc
        done       chan struct{}

        log *slog.Logger
}

func New(processors []Processor, sink Sink, opts ...Option) *Pipeline {
        stopCtx, stopCancel := context.WithCancel(context.Background())

        p := &Pipeline{
                eventsBufferSize: DefaultBufferSize,
                processors:       processors,
                sink:             sink,
                sinkTimeout:      DefaultSinkTimeout,
                workerCount:      DefaultWorkerCount,
                workerTimeout:    DefaultWorkerTimeout,
                stopCtx:          stopCtx,
                stopCancel:       stopCancel,
                done:             make(chan struct{}),
                log:              slog.New(slog.DiscardHandler),
        }

        for _, opt := range opts {
                opt(p)
        }

        p.events = make(chan event.Event, p.eventsBufferSize)

        return p
}

func (p *Pipeline) Ingest(ev event.Event) error {
        p.mu.RLock()
        defer p.mu.RUnlock()
        if p.closed {
                return ErrShuttingDown
        }
        select {
        case p.events <- ev:
                return nil
        default:
                return fmt.Errorf("ingest event: %w", ErrChanIsFull)
        }
}

func (p *Pipeline) Start() {
        defer func() {
                p.stopCancel()
                close(p.done)

                p.log.Info("pipeline stopped")
        }()

        p.log.Info("pipeline started")

        p.runWorkers()
}

func (p *Pipeline) Shutdown(ctx context.Context) error {
        p.mu.Lock()
        if p.closed {
                p.mu.Unlock()
                return ErrAlreadyStopped
        }
        close(p.events)
        p.closed = true
        p.mu.Unlock()

        select {
        case <-ctx.Done():
                p.stopCancel()
                return ctx.Err()
        case <-p.done:
                return nil
        }
}

func (p *Pipeline) runWorkers() {
        var wg sync.WaitGroup
        for range p.workerCount {
                wg.Go(func() {
                        for {
                                select {

                                case <-p.stopCtx.Done():
                                        if n := len(p.events); n > 0 {
                                                p.log.Warn("events left in pipeline", slog.Int("count", n))
                                        }
                                        return

                                case ev, ok := <-p.events:
                                        if !ok {
                                                p.log.Info("events drained")
                                                return
                                        }

                                        if p.stopCtx.Err() != nil {
                                                p.log.Warn("events left in pipeline", slog.Int("count", len(p.events)+1))
                                                return
                                        }

                                        procCtx, cancel := context.WithTimeout(p.stopCtx, p.workerTimeout)
                                        drop := p.runProcessors(procCtx, &ev)
                                        cancel()

                                        if !drop {
                                                sinkCtx, cancel := context.WithTimeout(p.stopCtx, p.sinkTimeout)
                                                err := p.sink(sinkCtx, ev)
                                                cancel()
                                                if err != nil {
                                                        p.log.Error(
                                                                "send processed event",
                                                                slog.Any("event", ev),
                                                                slog.String("error", err.Error()),
                                                        )
                                                }
                                        }

                                }
                        }
                })
        }

        wg.Wait()
}

func (p *Pipeline) runProcessors(ctx context.Context, ev *event.Event) bool {
        var drop bool

        for _, processor := range p.processors {
                var err error
                drop, err = processor.Process(ctx, ev)
                if err != nil {
                        p.log.Error(
                                "process event",
                                slog.Any("event", ev),
                                slog.String("error", err.Error()),
                        )
                        drop = true
                }
                if drop {
                        return drop
                }
        }
        return false
}

const (
        ResponseErrorInternalError  = "INTERNAL_ERROR"
        ResponseErrorBadRequestBody = "BAD_REQUEST_BODY"
        ResponseErrorBadEventID     = "BAD_EVENT_ID"
        ResponseErrorChanIsFull     = "CHAN_IS_FULL"
        ResponseErrorShuttingDown   = "SHUTTING_DOWN"
)

type eventJSON struct {
        ID         string          `json:"id"`
        Name       string          `json:"name"`
        UserID     string          `json:"user_id"`
        ClientTime time.Time       `json:"client_time"`
        Properties json.RawMessage `json:"properties"`
}

type errorResponseJSON struct {
        Error   string
        Message string
}

type Handler struct {
        ingester EventIngester

        log *slog.Logger
}

func NewHandler(ingester EventIngester, log *slog.Logger) *Handler {
        return &Handler{
                ingester: ingester,
                log:      log,
        }
}

func (h *Handler) HandleEvent(w http.ResponseWriter, req *http.Request) {
        var reqEvent eventJSON
        if err := json.NewDecoder(req.Body).Decode(&reqEvent); err != nil {
                h.writeError(w, http.StatusBadRequest, ResponseErrorBadRequestBody, "unable to parse request body: event json")
                return
        }
        evID, err := uuid.Parse(reqEvent.ID)
        if err != nil {
                h.writeError(w, http.StatusBadRequest, ResponseErrorBadEventID, "unable to parse request event id as uuid")
                return
        }

        ev := event.Event{
                ID:         evID,
                Name:       reqEvent.Name,
                UserID:     reqEvent.UserID,
                ClientTime: reqEvent.ClientTime,
                ServerTime: time.Now(),
                IP:         getIP(req),
                UserAgent:  req.Header.Get("User-Agent"),
                Properties: reqEvent.Properties,
        }

        if err := h.ingester.Ingest(ev); err != nil {
                code, error, message := mapError(err)
                if code == http.StatusInternalServerError {
                        h.log.Error("ingest event", slog.String("error", err.Error()))
                }
                h.writeError(w, code, error, message)
                return
        }

        w.WriteHeader(http.StatusOK)
}

func (h *Handler) writeError(w http.ResponseWriter, code int, error, message string) {
        w.WriteHeader(code)
        resp := errorResponseJSON{
                Error:   error,
                Message: message,
        }
        if err := json.NewEncoder(w).Encode(resp); err != nil {
                h.log.Error(
                        "marshalling error response",
                        slog.String("error", err.Error()),
                        slog.Group(
                                "response",
                                slog.String("error", error),
                                slog.String("message", message),
                        ),
                )
        }
}

func getIP(req *http.Request) string {
        if header := req.Header.Get("X-Forwarded-For"); header != "" {
                ips := strings.Split(header, ", ")
                return ips[0]
        }
        return req.RemoteAddr
}

func mapError(err error) (code int, error string, message string) {
        switch {
        case errors.Is(err, pipeline.ErrChanIsFull):
                return http.StatusTooManyRequests, ResponseErrorChanIsFull, "events queue is full, try again later"
        case errors.Is(err, pipeline.ErrShuttingDown):
                return http.StatusServiceUnavailable, ResponseErrorShuttingDown, "service is shutting down"
        default:
                return http.StatusInternalServerError, ResponseErrorInternalError, "internal server error"
        }
}

package server

import (
        "context"
        "errors"
        "log/slog"
        "net/http"

        "github.com/neelalala/hopper/internal/event"
)

type EventIngester interface {
        Ingest(event.Event) error
}

type Server struct {
        httpServer *http.Server

        ingester EventIngester

        log *slog.Logger
}

func New(addr string, ingester EventIngester, log *slog.Logger) *Server {
        handler := NewHandler(ingester, log.With(slog.String("handler", "events")))

        mux := http.NewServeMux()

        mux.HandleFunc("POST /events", handler.HandleEvent)

        httpServer := &http.Server{
                Addr:    addr,
                Handler: mux,
        }

        server := &Server{
                httpServer: httpServer,
                ingester:   ingester,
                log:        log,
        }

        return server
}

func (s *Server) Start() error {
        defer s.log.Info("server stopped")
        s.log.Info("server satrted", slog.String("address", s.httpServer.Addr))
        if err := s.httpServer.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
                return err
        }
        return nil
}

func (s *Server) Shutdown(ctx context.Context) error {
        defer s.log.Info("server shutted down")
        s.log.Info("shutting down server")
        return s.httpServer.Shutdown(ctx)
}
